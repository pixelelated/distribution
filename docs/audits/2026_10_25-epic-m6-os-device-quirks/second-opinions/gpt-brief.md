## 3.1 Architectural coherence

- **Selection is by name, and the name is the device tree's.**
  - `QUIRK_DEVICE` is the dt `model` (or the DMI pair) with `/` → `-` (`002-autostart:6-12`), and a directory under `devices/` is selected when its name equals it.

- **Two kinds of file, two ways in.**

 3  # SPDX-License-Identifier: GPL-2.0-or-later

18  G=$3
19  B=$4
20

23    if ! [ "$val" -ge 0 ] || ! [ "$val" -le 255 ]; then
24      echo "RGB values must be 0-255"
25      exit 1
26    fi
27  done
28  
29  # Convert to hex
30  BRIGHTNESS_HEX=$(printf "%02X" $BRIGHTNESS)

32  G_HEX=$(printf "%02X" $G)
33  B_HEX=$(printf "%02X" $B)
34  
35  # Function to calculate checksum
36  calculate_checksum() {
37    local sum=0
38    for byte in "$@"; do
39      sum=$((sum + byte))
40    done
41    echo $((sum & 0xFF))
42  }
43  
44  # Open serial device
45  exec 20<>$SERIAL_DEVICE
46  stty -F $SERIAL_DEVICE 115200 -opost -isig -icanon -echo
47  
48  # Construct payload (without STX and checksum)
49  payload=(0x00 0x1C 0x20 0x01 0x80 0x00 0x81 0x00 0x8B 0x0F 0x88 0x$R_HEX 0x89 0x$G_HEX 0x8A 0x$B_HEX 0x86 0xFF 0x87 0x$BRIGHTNESS_HEX 0x58 0x02 0x45 0x00)
50  
51  # Calculate checksum
52  CHECKSUM=$(calculate_checksum "${payload[@]}")
53  
54  # Construct full frame: STX + payload + checksum + ETX
55  STX=0xF7
56  ETX=0xED
57  FRAME=($STX "${payload[@]}" $CHECKSUM $ETX)
58  
59  # Convert to \x.. string
60  CMD=""
61  for b in "${FRAME[@]}"; do
62    CMD+="\\x$(printf '%02X' $b)"
63  done
64  
65  # Send frame
66  echo -ne "$CMD" > $SERIAL_DEVICE
67
