+# RCLONE_NET_OPTS_ARRAY from <the conf's value>, through an allowlist
+# (D-CLOUD-185, #1129). The value is a player-editable string, and it was
+# split whole into every transfer, the match's deleting sync among them:
+# --delete-excluded there deleted the saves beside the ROMs, and so did an
+# --include ahead of the save excludes (T17's F-02, Argon's R-9). Only the
+# bound's flags pass, each with a value of its own kind -- a duration in the
+# shape idle_timeout_seconds reads, a count, a rate -- as `--flag value` or
+# `--flag=value`. Anything else is left out, and one log line names what by
+# the flag's name alone, never a value. Nothing left means the shipped bound
+# (<fallback>), as an empty line always has.
+content_net_opts() {
+    local words w flag val ok dropped="" i=0
+    read -r -a words <<< "$1"
+    RCLONE_NET_OPTS_ARRAY=()
+    while [ "${i}" -lt "${#words[@]}" ]; do
+        w="${words[${i}]}"; i=$((i + 1))
+        case "${w}" in
+            --*=*) flag="${w%%=*}"; val="${w#*=}" ;;
+            --*)   flag="${w}"; val="" ;;
+            *)     dropped="${dropped} (a word that is not a flag)"; continue ;;
+        esac
+        case "${flag}" in
+            --contimeout|--timeout|--low-level-retries|--retries|--bwlimit)
+                if [ "${w}" = "${flag}" ] && [ "${i}" -lt "${#words[@]}" ] && [ "${words[${i}]#--}" = "${words[${i}]}" ]; then
+                    val="${words[${i}]}"; i=$((i + 1))
+                fi ;;
+            *)
+                dropped="${dropped} $(printf '%s' "${flag}" | tr -c 'A-Za-z0-9-' '?' | cut -c1-40)"
+                continue ;;
+        esac
+        case "${flag}" in
+            --contimeout|--timeout)          [[ "${val}" =~ ^[0-9]+(ms|s|m|h)?$ ]] ;;
+            --low-level-retries|--retries)   [[ "${val}" =~ ^[0-9]{1,3}$ ]] ;;
+            --bwlimit)                       [[ "${val}" =~ ^(off|[0-9]+(\.[0-9]+)?[KkMmGg]?(:[0-9]+(\.[0-9]+)?[KkMmGg]?)?)$ ]] ;;
+        esac
+        ok=$?
+        if [ "${ok}" -eq 0 ]; then
+            RCLONE_NET_OPTS_ARRAY+=("${flag}" "${val}")
+        else
+            dropped="${dropped} ${flag} (its value is not a ${flag#--} value)"
+        fi
+    done
+    [ -z "${dropped}" ] || log_message "RCLONE_NET_OPTS: left out, not on the allowlist (D-CLOUD-185):${dropped}"
+    if [ "${#RCLONE_NET_OPTS_ARRAY[@]}" -eq 0 ]; then
+        read -r -a RCLONE_NET_OPTS_ARRAY <<< "$2"
+    fi
