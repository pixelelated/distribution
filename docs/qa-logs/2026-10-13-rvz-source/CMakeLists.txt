# UNRUN standalone hash control. Use coordinator-approved compiler/sysroot.
cmake_minimum_required(VERSION 3.10)
project(pix574_hash_control C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_DIRECTORY "${ES_SOURCE}/external/libcheevos")
    message(FATAL_ERROR "Set ES_SOURCE to the exact approved ES source checkout")
endif()
add_subdirectory("${ES_SOURCE}/external/libcheevos" libcheevos)
add_executable(pix574-hash-control hash-control.cpp)
target_include_directories(pix574-hash-control PRIVATE "${ES_SOURCE}/external/libcheevos")
target_link_libraries(pix574-hash-control PRIVATE libcheevos)
if(UNIX)
    target_link_libraries(pix574-hash-control PRIVATE m)
endif()
